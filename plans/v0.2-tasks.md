class DouyinLauncher(private val context: Context) {
    private val DOUYIN_PACKAGE = "com.ss.android.ugc.aweme"
    
    fun isDouyinInstalled(): Boolean {
        return try {
            context.packageManager.getPackageInfo(DOUYIN_PACKAGE, 0)
            true
        } catch (e: Exception) {
            false
        }
    }
    
    fun launchDouyin(): Boolean {
        if (!isDouyinInstalled()) {
            return false
        }
        
        val intent = context.packageManager.getLaunchIntentForPackage(DOUYIN_PACKAGE)
        intent?.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        context.startActivity(intent)
        
        return true
    }
}
```

class ElementLocator(private val service: AccessibilityService) {
    
    fun findById(id: String, timeout: Long = 5000): AccessibilityNodeInfo? {
        val startTime = System.currentTimeMillis()
        while (System.currentTimeMillis() - startTime < timeout) {
            val root = service.rootInActiveWindow ?: continue
            val node = findNodeById(root, id)
            if (node != null) return node
            Thread.sleep(500)
        }
        return null
    }
    
    fun findByText(text: String, exact: Boolean = false): AccessibilityNodeInfo? {
        val root = service.rootInActiveWindow ?: return null
        return findNodeByText(root, text, exact)
    }
    
    private fun findNodeById(node: AccessibilityNodeInfo, id: String): AccessibilityNodeInfo? {
        if (node.viewIdResourceName == id) return node
        for (i in 0 until node.childCount) {
            val child = node.getChild(i) ?: continue
            val result = findNodeById(child, id)
            if (result != null) return result
        }
        return null
    }
}
```

class DouyinNavigator(
    private val service: AccessibilityService,
    private val locator: ElementLocator
) {
    suspend fun searchBlogger(name: String): Boolean {
        // 1. 点击搜索按钮
        if (!clickSearchButton()) {
            Log.e(TAG, "无法点击搜索按钮")
            return false
        }
        delay(1000)
        
        // 2. 输入博主名称
        if (!inputSearchText(name)) {
            Log.e(TAG, "无法输入搜索文本")
            return false
        }
        delay(500)
        
        // 3. 触发搜索
        if (!triggerSearch()) {
            Log.e(TAG, "无法触发搜索")
            return false
        }
        delay(2000)
        
        // 4. 等待结果
        return waitForResults()
    }
    
    private fun clickSearchButton(): Boolean {
        val node = locator.findByText(DouyinElements.SEARCH_BUTTON)
            ?: locator.findById(DouyinElements.SEARCH_ICON_ID)
        return node?.performAction(AccessibilityNodeInfo.ACTION_CLICK) ?: false
    }
    
    private fun inputSearchText(text: String): Boolean {
        val node = locator.findById(DouyinElements.SEARCH_INPUT, timeout = 3000)
            ?: return false
        
        // 先点击获取焦点
        node.performAction(AccessibilityNodeInfo.ACTION_CLICK)
        delay(300)
        
        // 输入文本
        val arguments = Bundle().apply {
            putCharSequence(
                AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE,
                text
            )
        }
        return node.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, arguments)
    }
}
```

class ResultParser(private val locator: ElementLocator) {
    
    fun parseSearchResults(): List<SearchResult> {
        val results = mutableListOf<SearchResult>()
        val listNode = locator.findById(DouyinElements.RESULT_LIST)
            ?: return results
        
        for (i in 0 until listNode.childCount) {
            val itemNode = listNode.getChild(i) ?: continue
            val result = parseResultItem(itemNode)
            if (result != null) {
                results.add(result)
            }
        }
        
        return results
    }
    
    private fun parseResultItem(node: AccessibilityNodeInfo): SearchResult? {
        val nickname = findTextInNode(node, "nickname") ?: return null
        val douyinId = findTextInNode(node, "douyin_id")
        val fansCount = findTextInNode(node, "fans")
        
        return SearchResult(
            nickname = nickname,
            douyinId = douyinId,
            fansCount = fansCount,
            isVerified = checkVerified(node)
        )
    }
}

    val step: String,
    val action: String,

)

class OperationLogger(private val wsClient: WebSocketClient) {
    private val logs = mutableListOf<OperationLog>()
    
    fun log(step: String, action: String, result: Boolean, message: String? = null) {
        val log = OperationLog(
            timestamp = System.currentTimeMillis(),
            step = step,
            action = action,
            result = result,
            message = message,
            screenshot = if (shouldTakeScreenshot(step)) takeScreenshot() else null
        )
        
        logs.add(log)
        reportLog(log)
    }
    
    private fun reportLog(log: OperationLog) {
        val message = json.dumps({
            "type": "operation_log",
            "log": log
        })
        wsClient.sendMessage(message)
    }
    
    fun getAllLogs(): List<OperationLog> = logs.toList()
}
```

```kotlin
class SearchBloggerTask(
    private val service: AutomationAccessibilityService,
    private val wsClient: WebSocketClient
) {
    private val launcher = DouyinLauncher(service)
    private val locator = ElementLocator(service)
    private val navigator = DouyinNavigator(service, locator)
    private val profileNav = ProfileNavigator(service, locator)
    private val logger = OperationLogger(wsClient)
    
    suspend fun execute(bloggerName: String): Boolean {
        try {
            logger.log("开始", "搜索博主: $bloggerName", true)
            
            // 1. 启动抖音
            if (!launcher.launchDouyin()) {
                logger.log("启动抖音", "失败", false, "抖音未安装或无法启动")
                return false
            }
            logger.log("启动抖音", "成功", true)
            delay(3000)
            
            // 2. 搜索博主
            if (!navigator.searchBlogger(bloggerName)) {
                logger.log("搜索博主", "失败", false, "搜索失败")
                return false
            }
            logger.log("搜索博主", "成功", true)
            
            // 3. 进入主页
            if (!profileNav.enterProfile(bloggerName)) {
                logger.log("进入主页", "失败", false, "无法进入主页")
                return false
            }
            logger.log("进入主页", "成功", true)
            
            // 4. 采集信息
            val profileInfo = profileNav.getProfileInfo()
            logger.log("采集信息", "成功", true)
            
            // 5. 上报结果
            reportSuccess(bloggerName, profileInfo)
            
            return true
            
        } catch (e: Exception) {
            logger.log("执行任务", "异常", false, e.message)
            return false
        }
    }
    
    private fun reportSuccess(bloggerName: String, info: ProfileInfo) {
        val message = json.dumps({
            "type": "search_result",
            "success": true,
            "blogger_name": bloggerName,
            "profile_info": info,
            "logs": logger.getAllLogs()
        })
        wsClient.sendMessage(message)
    }
}
```
