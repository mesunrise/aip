# v0.2 版本需求文档

## 版本信息

**版本号：** v0.2  
**版本名称：** 博主定位  
**开发周期：** 预计5-7天  
**依赖版本：** v0.1（通信功能）

## 版本目标

实现抖音App的启动、博主搜索和主页定位功能，为后续的评论采集提供基础。

## 核心功能流程

```
云端发送指令 → App启动抖音 → 搜索博主 → 进入主页 → 上报结果
     ↓              ↓            ↓          ↓          ↓
  "搜索XXX"    打开抖音App   输入博主名   点击结果   返回成功
```

### 详细流程

```
1. 云端控制台
   ├─ 输入博主名称："张三"
   ├─ 点击"搜索博主"按钮
   └─ 发送WebSocket消息
       {
         "type": "search_blogger",
         "blogger_name": "张三"
       }

2. App接收指令
   ├─ 解析消息
   ├─ 验证抖音是否安装
   └─ 启动自动化流程

3. 启动抖音
   ├─ 使用Intent启动抖音
   ├─ 等待抖音界面加载
   └─ 检测是否成功启动

4. 搜索博主
   ├─ 定位搜索按钮
   ├─ 点击搜索按钮
   ├─ 定位搜索输入框
   ├─ 输入博主名称
   ├─ 点击搜索/确认
   └─ 等待搜索结果

5. 进入博主主页
   ├─ 定位搜索结果列表
   ├─ 识别博主账号
   ├─ 点击博主头像/名称
   ├─ 等待主页加载
   └─ 验证是否进入主页

6. 上报结果
   ├─ 采集主页信息
   ├─ 截图保存
   └─ 发送WebSocket消息
       {
         "type": "search_result",
         "success": true,
         "blogger_name": "张三",
         "blogger_id": "xxx",
         "screenshot": "base64..."
       }
```

class DouyinLauncher {
    fun launchDouyin(): Boolean {
        val packageName = "com.ss.android.ugc.aweme"
        
        // 检查是否安装
        if (!isAppInstalled(packageName)) {
            return false
        }
        
        // 启动抖音
        val intent = packageManager.getLaunchIntentForPackage(packageName)
        startActivity(intent)
        
        // 等待加载
        delay(3000)
        
        return isDouyinRunning()
    }
}

```kotlin
class DouyinSearcher(private val service: AccessibilityService) {
    
    suspend fun searchBlogger(name: String): Boolean {
        // 1. 点击搜索按钮
        if (!clickSearchButton()) {
            return false
        }
        delay(1000)
        
        // 2. 输入博主名称
        if (!inputSearchText(name)) {
            return false
        }
        delay(500)
        
        // 3. 触发搜索
        if (!triggerSearch()) {
            return false
        }
        delay(2000)
        
        // 4. 等待结果
        return waitForResults()
    }
    
    private fun clickSearchButton(): Boolean {
        val node = findElementByText("搜索") 
            ?: findElementById(SEARCH_BUTTON_ID)
        return node?.performAction(AccessibilityNodeInfo.ACTION_CLICK) ?: false
    }
    
    private fun inputSearchText(text: String): Boolean {
        val node = findElementById(SEARCH_INPUT_ID)
        val arguments = Bundle().apply {
            putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, text)
        }
        return node?.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, arguments) ?: false
    }
}
```

class ResultParser {
    fun parseSearchResults(): List<SearchResult> {
        val results = mutableListOf<SearchResult>()
        val listNode = findElementById(RESULT_LIST_ID) ?: return results
        
        // 遍历结果列表
        for (i in 0 until listNode.childCount) {
            val itemNode = listNode.getChild(i)
            val result = parseResultItem(itemNode)
            if (result != null) {
                results.add(result)
            }
        }
        
        return results
    }
    
    private fun parseResultItem(node: AccessibilityNodeInfo): SearchResult? {
        // 提取昵称、抖音号等信息
        val nickname = findTextByType(node, "nickname")
        val douyinId = findTextByType(node, "douyin_id")
        // ...
        return SearchResult(nickname, douyinId, ...)
    }
}

class OperationLogger {
    private val logs = mutableListOf<OperationLog>()
    
    fun log(step: String, action: String, result: Boolean, message: String? = null) {
        val log = OperationLog(
            timestamp = System.currentTimeMillis(),
            step = step,
            action = action,
            result = result,

    private fun shouldTakeScreenshot(step: String): Boolean {
        return step in listOf("启动抖音", "搜索结果", "进入主页")
    }
}

class ScreenshotUtil {

#### 2.2 服务实现
```kotlin
class AutomationAccessibilityService : AccessibilityService() {
    
    private lateinit var douyinNavigator: DouyinNavigator
    private lateinit var operationLogger: OperationLogger
    
    override fun onServiceConnected() {
        super.onServiceConnected()
        Log.d(TAG, "无障碍服务已连接")
        
        douyinNavigator = DouyinNavigator(this)
        operationLogger = OperationLogger()
        
        // 通知MainActivity服务已启动
        sendBroadcast(Intent(ACTION_SERVICE_CONNECTED))
    }
    
    override fun onAccessibilityEvent(event: AccessibilityEvent) {
        when (event.eventType) {
            AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED -> {
                handleWindowStateChanged(event)
            }
            AccessibilityEvent.TYPE_WINDOW_CONTENT_CHANGED -> {
                handleContentChanged(event)
            }
        }
    }
    
    override fun onInterrupt() {
        Log.d(TAG, "无障碍服务中断")
    }
    
    private fun handleWindowStateChanged(event: AccessibilityEvent) {
        val packageName = event.packageName?.toString()
        val className = event.className?.toString()
        
        Log.d(TAG, "窗口变化: $packageName - $className")
        
        if (packageName == "com.ss.android.ugc.aweme") {
            // 抖音界面变化
            douyinNavigator.onWindowChanged(className)
        }
    }
    
    // 执行搜索任务
    fun executeSearchTask(bloggerName: String) {
        GlobalScope.launch {
            try {
                operationLogger.log("开始", "搜索博主: $bloggerName", true)
                
                // 1. 启动抖音
                if (!launchDouyin()) {
                    operationLogger.log("启动抖音", "失败", false, "无法启动抖音")
                    return@launch
                }
                operationLogger.log("启动抖音", "成功", true)
                
                // 2. 搜索博主
                if (!douyinNavigator.searchBlogger(bloggerName)) {
                    operationLogger.log("搜索博主", "失败", false, "搜索失败")
                    return@launch
                }
                operationLogger.log("搜索博主", "成功", true)
                
                // 3. 进入主页
                if (!douyinNavigator.enterProfile()) {
                    operationLogger.log("进入主页", "失败", false, "无法进入主页")
                    return@launch
                }
                operationLogger.log("进入主页", "成功", true)
                
                // 4. 上报结果
                reportSuccess(bloggerName)
                
            } catch (e: Exception) {
                operationLogger.log("执行任务", "异常", false, e.message)
            }
        }
    }
}
```

### 3. 云端功能

#### 3.1 搜索指令下发
```python
@app.post("/api/search_blogger")
async def search_blogger(blogger_name: str, device_id: str):
    """发送搜索博主指令"""
    if device_id not in connected_devices:
        return {"success": False, "message": "设备未连接"}
    
    websocket = connected_devices[device_id]
    
    # 发送搜索指令
    await websocket.send_text(json.dumps({
        "type": "search_blogger",
        "blogger_name": blogger_name,
        "timestamp": datetime.now().isoformat()
    }))
    
    print(f"📤 发送搜索指令 [{device_id}]: {blogger_name}")
    
    return {"success": True, "message": "指令已发送"}
```
